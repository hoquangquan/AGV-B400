\section*{DANH MỤC KÝ HIỆU VÀ CHỮ VIẾT TẮT}
\addcontentsline{toc}{section}{\numberline{}DANH MỤC KÝ HIỆU VÀ CHỮ VIẾT TẮT}

\begin{itemize}
    \item \textbf{AGV}: Automated Guided Vehicle (Xe tự hành dẫn đường tự động trong nhà máy)
    \item \textbf{MCU}: Microcontroller Unit (Bộ vi điều khiển trung tâm)
    \item \textbf{RTOS}: Real-Time Operating System (Hệ điều hành thời gian thực)
    \item \textbf{CAN}: Controller Area Network (Chuẩn mạng truyền thông vùng điều khiển công nghiệp)
    \item \textbf{MGS}: Magnetic Guide Sensor (Cảm biến từ dò đường dán trên sàn)
    \item \textbf{RFID}: Radio Frequency Identification (Hệ thống định vị thẻ từ nhận dạng tần số vô tuyến)
    \item \textbf{BMS}: Battery Management System (Hệ thống quản lý và giám sát khối pin Lithium)
    \item \textbf{HMI}: Human-Machine Interface (Màn hình giao diện tương tác Người - Máy)
    \item \textbf{PLC}: Programmable Logic Controller (Bộ điều khiển logic lập trình được tại trạm hàng)
    \item \textbf{ADC}: Analog-to-Digital Converter (Bộ chuyển đổi tín hiệu tương tự sang số)
    \item \textbf{DMA}: Direct Memory Access (Cơ chế truy cập bộ nhớ trực tiếp không qua CPU)
    \item \textbf{EMC}: Emergency Stop (Trạng thái ngắt dừng khẩn cấp an toàn)
    \item \textbf{Lidar}: Light Detection and Ranging (Cảm biến quét vật cản và đo khoảng cách bằng laser)
\end{itemize}
